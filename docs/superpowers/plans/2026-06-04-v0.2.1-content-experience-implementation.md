- [ ] **Step 3: 创建 helper 实现**

Create `src/lib/reviews/display.ts`:

```ts
import type { PortableBlock, Review, ReviewCategory, ReviewCover } from '../content/types';

export const REVIEW_DEFAULT_COVERS: Record<ReviewCategory, ReviewCover> = {
  music: { src: '/covers/reviews/default-music.png', alt: '音乐默认封面' },
  book: { src: '/covers/reviews/default-book.png', alt: '书籍默认封面' },
  film: { src: '/covers/reviews/default-film.png', alt: '影视默认封面' },
};

export function clampReviewRating(value: number): number {
  if (!Number.isFinite(value)) return 0;
  return Math.min(5, Math.max(0, Math.round(value * 10) / 10));
}

export function formatReviewRating(value: number): string {
  return `${clampReviewRating(value).toFixed(1)}/5`;
}

export function getStarSegments(value: number): number[] {
  const rating = clampReviewRating(value);

  return Array.from({ length: 5 }, (_, index) => {
    const remaining = rating - index;
    if (remaining >= 1) return 100;
    if (remaining <= 0) return 0;
    return Math.round(remaining * 10) * 10;
  });
}

export function getDefaultReviewCover(category: ReviewCategory): ReviewCover {
  return REVIEW_DEFAULT_COVERS[category];
}

export function getReviewCover(review: Review): ReviewCover {
  return review.cover ?? getDefaultReviewCover(review.category);
}

export function hasLongReview(review: Pick<Review, 'body'>): boolean {
  return review.body.some((block: PortableBlock) => block._type === 'block');
}

export function formatReviewDate(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `记录于 ${year}.${month}.${day}`;
}
```

```ts
export function mapSanityReview(doc: SanityReviewDocument): Review {
  const title = text(doc.title);
  const category = doc.category ?? 'music';
  const imageUrl = getSanityImageUrl(doc.cover);
  const fallbackCover = getDefaultReviewCover(category);

  return {
    title,
    slug: text(doc.slug),
    category,
    creator: text(doc.creator),
    year: typeof doc.year === 'number' ? doc.year : 0,
    date: dateOrEpoch(doc.date),
    rating: clampReviewRating(typeof doc.rating === 'number' ? doc.rating : 0),
    cover: {
      src: imageUrl ?? fallbackCover.src,
      alt: imageUrl ? `${title} 封面` : fallbackCover.alt,
    },
    summary: text(doc.summary),
    tags: stringArray(doc.tags),
    moments: stringArray(doc.moments),
    body: portableBody(doc.body),
  };
}
```

```astro
<script>
  const yearButtons = document.querySelectorAll('[data-filter-year]');
  const tagButtons = document.querySelectorAll('[data-filter-tag]');
  const entries = Array.from(document.querySelectorAll<HTMLElement>('[data-year][data-tags][data-page-item]'));
  const pagination = document.querySelector<HTMLElement>('[data-pagination]');
  const emptyFilter = document.querySelector<HTMLElement>('[data-empty-filter]');
  const pageSize = Number(pagination?.dataset.pageSize ?? 6);
  const prev = pagination?.querySelector<HTMLButtonElement>('[data-page-prev]');
  const next = pagination?.querySelector<HTMLButtonElement>('[data-page-next]');
  const status = pagination?.querySelector<HTMLElement>('[data-page-status]');
  let selectedYear = 'all';
  let selectedTag = 'all';
  let page = 1;

  function updateActive(buttons: NodeListOf<Element>, selectedValue: string, attribute: string) {
    buttons.forEach((button) => {
      button.classList.toggle('is-active', button.getAttribute(attribute) === selectedValue);
    });
  }

  function getFilteredEntries() {
    return entries.filter((entry) => {
      const yearMatches = selectedYear === 'all' || entry.getAttribute('data-year') === selectedYear;
      const tagList = (entry.getAttribute('data-tags') ?? '').split(',');
      const tagMatches = selectedTag === 'all' || tagList.includes(selectedTag);
      return yearMatches && tagMatches;
    });
  }

  function render() {
    const filteredEntries = getFilteredEntries();
    const pageCount = Math.max(1, Math.ceil(filteredEntries.length / pageSize));
    page = Math.min(page, pageCount);

    entries.forEach((entry) => entry.toggleAttribute('hidden', true));
    filteredEntries.forEach((entry, index) => {
      const visible = index >= (page - 1) * pageSize && index < page * pageSize;
      entry.toggleAttribute('hidden', !visible);
    });

    if (emptyFilter) emptyFilter.hidden = filteredEntries.length > 0;
    if (pagination) pagination.hidden = filteredEntries.length <= pageSize;
    if (status) status.textContent = `第 ${page} / ${pageCount} 页`;
    if (prev) prev.disabled = page <= 1;
    if (next) next.disabled = page >= pageCount;

    updateActive(yearButtons, selectedYear, 'data-filter-year');
    updateActive(tagButtons, selectedTag, 'data-filter-tag');
  }

  yearButtons.forEach((button) => {
    button.addEventListener('click', () => {
      selectedYear = button.getAttribute('data-filter-year') ?? 'all';
      page = 1;
      render();
    });
  });

  tagButtons.forEach((button) => {
    button.addEventListener('click', () => {
      selectedTag = button.getAttribute('data-filter-tag') ?? 'all';
      page = 1;
      render();
    });
  });

  prev?.addEventListener('click', () => {
    page -= 1;
    render();
  });

  next?.addEventListener('click', () => {
    page += 1;
    render();
  });

  render();
</script>
```
